**Goal:** 交付 Adapter 的协议层（A：invocations/ws/a2a/agui handlers + main 装配）与并发控制（B：session-queue + run），使镜像主进程具备三端点服务能力，a2a/agui 按约定契约就绪待 SDK 接线。

```ts
/**
 * 帧契约（spec §3.1）：driver 写入 stdout 的 NDJSON，一行一帧。
 * 帧类型命名不带 DSH 词汇——这是跨 harness 的公共契约。
 */

export interface SessionFrame { type: 'session'; sessionId: string; cwd: string }
export interface StatusFrame {
    type: 'status'
    phase: 'turn_start' | 'step_start' | 'step_end' | 'turn_end'
}
export interface DeltaFrame {
    type: 'delta'; kind: string; text: string; attemptId: string; revision: number
}
export interface ToolCallFrame { type: 'tool_call'; callId: string; tool: string; input: unknown }
export interface ToolResultFrame {
    type: 'tool_result'; callId: string; status: string; result: unknown
}
export interface QuestionFrame {
    type: 'question'; callId: string; questions: unknown[]; timeoutSec: number
}
export interface FinalFrame {
    type: 'final'; text: string; reason: { kind: string }; sessionId: string
}
export interface ErrorFrame { type: 'error'; message: string }

export type Frame =
    | SessionFrame | StatusFrame | DeltaFrame | ToolCallFrame
    | ToolResultFrame | QuestionFrame | FinalFrame | ErrorFrame

export type ParseResult =
    | { ok: true; frame: Frame }
    | { ok: false; reason: 'empty' | 'invalid'; line: string }

const FRAME_TYPES = new Set([
    'session', 'status', 'delta', 'tool_call', 'tool_result', 'question', 'final', 'error',
])

/** 解析一行 NDJSON。只校验 type 合法——字段级校验由产帧方（driver）负责。 */
export function parseFrame(line: string): ParseResult {
    if (line.trim() === '') return { ok: false, reason: 'empty', line }
    let value: unknown
    try {
        value = JSON.parse(line)
    } catch {
        return { ok: false, reason: 'invalid', line }
    }
    if (typeof value !== 'object' || value === null) return { ok: false, reason: 'invalid', line }
    const type = (value as { type?: unknown }).type
    if (typeof type !== 'string' || !FRAME_TYPES.has(type)) {
        return { ok: false, reason: 'invalid', line }
    }
    return { ok: true, frame: value as Frame }
}
```

- [ ] **Step 4: 实现 `adapter/src/scheduler/session-queue.ts`**

```ts
/**
 * 同 sessionId 串行队列（spec §4.4）：一个会话同一时刻只由一个子进程持有。
 * 全局并发上限不在这里——由 SDK maxConcurrency 执行。
 */
export class SessionQueue {
    /** 每个会话的队尾 Promise；新 job 接在队尾之后 */
    private tails = new Map<string, Promise<unknown>>()

    /** 当前排队/执行中的会话数（测试与观测用） */
    size(): number {
        return this.tails.size
    }

    enqueue<T>(sessionId: string, job: () => Promise<T>): Promise<T> {
        const prev = this.tails.get(sessionId) ?? Promise.resolve()
        // 前序成功或失败都接着跑（失败不阻塞会话后续请求）
        const run = prev.then(job, job)
        // 队尾只关心"完成"，不携带结果与错误，避免未处理 rejection
        const tail = run.then(() => undefined, () => undefined)
        this.tails.set(sessionId, tail)
        // 队尾结算时若没有更新的 job 接上，则清理条目
        void tail.then(() => {
            if (this.tails.get(sessionId) === tail) this.tails.delete(sessionId)
        })
        return run
    }
}
```

```ts
import { spawn } from 'node:child_process'
import { createInterface } from 'node:readline'
import {
    parseFrame, type Frame, type FinalFrame, type LaunchSpec,
} from '@agentarts/harness-dsh-shared'

/** 一轮超时默认值（spec §7 第 8 项待与平台对齐，先按 5 分钟） */
export const DEFAULT_ROUND_TIMEOUT_MS = 300_000

export interface RoundOutcome {
    ok: boolean
    final: FinalFrame | null
    error: string | null
    exitCode: number | null
}

export interface RoundHandle {
    /** 帧流，单消费者；子进程结束且缓冲 drain 后迭代完成 */
    frames: AsyncIterable<Frame>
    /** 子进程退出后结算 */
    outcome: Promise<RoundOutcome>
}

/** 简易帧通道：生产端 push/end，消费端异步迭代（单消费者，无背压需求——帧量小） */
function createFrameChannel() {
    const queue: Frame[] = []
    let notify: (() => void) | null = null
    let ended = false
    return {
        push(frame: Frame) { queue.push(frame); notify?.() },
        end() { ended = true; notify?.() },
        frames: {
            async *[Symbol.asyncIterator]() {
                while (true) {
                    while (queue.length > 0) yield queue.shift()!
                    if (ended) return
                    // 检查与注册在同一同步段内完成，不会丢失唤醒
                    await new Promise<void>((resolve) => { notify = resolve })
                    notify = null
                }
            },
        } as AsyncIterable<Frame>,
    }
}

/**
 * 执行一轮（spec §3.2）：spawn 子进程，stdin 写任务，stdout 逐行解析帧，
 * 按退出码与 final 帧判成败。B 不缓存结果——幂等判重在子进程内（D）。
 */
export function runRound(
    spec: LaunchSpec, task: string, timeoutMs = DEFAULT_ROUND_TIMEOUT_MS,
): RoundHandle {
    const channel = createFrameChannel()
    let final: FinalFrame | null = null
    let stderrTail = ''
    let failReason: string | null = null

    const child = spawn(spec.command, spec.args, {
        env: spec.env, stdio: ['pipe', 'pipe', 'pipe'],
    })

    const outcome = new Promise<RoundOutcome>((resolve) => {
        const timer = setTimeout(() => {
            failReason = `round timeout after ${timeoutMs}ms`
            child.kill('SIGKILL')
        }, timeoutMs)

        child.on('error', (err) => {
            clearTimeout(timer)
            failReason = `spawn failed: ${err.message}`
            channel.end()
            resolve({ ok: false, final: null, error: failReason, exitCode: null })
        })

        const lines = createInterface({ input: child.stdout })
        lines.on('line', (line) => {
            const r = parseFrame(line)
            if (r.ok) {
                if (r.frame.type === 'final') final = r.frame
                channel.push(r.frame)
            } else if (r.reason === 'invalid') {
                // 协议被破坏：立即判失败并终止子进程
                failReason = `invalid frame line: ${line.slice(0, 200)}`
                child.kill('SIGKILL')
            }
        })

        child.stderr.on('data', (chunk) => {
            stderrTail = (stderrTail + String(chunk)).slice(-8192)
        })

        child.on('close', (code) => {
            clearTimeout(timer)
            channel.end()
            const ok = code === 0 && final !== null && failReason === null
            const error = ok ? null
                : failReason
                    ?? (code !== 0 ? `exit ${code}: ${stderrTail}` : 'no final frame')
            resolve({ ok, final, error, exitCode: code })
        })

        // 子进程早退时 stdin 可能 EPIPE，忽略（成败由 close/error 判定）
        child.stdin.on('error', () => {})
        child.stdin.end(spec.encodeStdin(task))
    })

    return { frames: channel.frames, outcome }
}
```

```ts
import type { RoundHandle } from './run.js'
import { runRound } from './run.js'
import { SessionQueue } from './session-queue.js'
import type { SchedulerDeps } from '../types.js'
import type { RoundRequest } from '@agentarts/harness-dsh-shared'

/**
 * B 模块门面：同 sessionId 串行（SessionQueue）+ 每轮一个子进程（runRound）。
 * 全局并发上限由 SDK maxConcurrency 执行，不在本层。
 */
export class Scheduler {
    private queue = new SessionQueue()

    constructor(private deps: SchedulerDeps) {}

    run(req: RoundRequest): Promise<RoundHandle> {
        return new Promise<RoundHandle>((resolve, reject) => {
            void this.queue.enqueue(req.sessionId, async () => {
                // 召回失败降级为空（spec §2.2 第 3 步），不阻塞执行
                let recallResult = ''
                try {
                    recallResult = (await this.deps.recall?.(req)) ?? ''
                } catch {
                    recallResult = ''
                }
                const spec = await this.deps.launchSpec(req, recallResult)
                const handle = runRound(spec, req.prompt, this.deps.timeoutMs)
                // 先把 handle 交给调用方（A 要边跑边消费帧流）……
                resolve(handle)
                // ……但队列名额必须等本轮结束才释放，否则同会话会并行两个子进程
                await handle.outcome
            }).catch(reject)
        })
    }
}
```

```ts
import { randomUUID } from 'node:crypto'
import type { RoundRequest } from '@agentarts/harness-dsh-shared'

function asString(v: unknown): string | null {
    return typeof v === 'string' && v !== '' ? v : null
}

/** payload → 四元组。sessionId 优先级：payload > 请求头 > 生成（spec §2.2）。 */
export function normalizeRequest(payload: any, headerSessionId?: string | null): RoundRequest {
    if (typeof payload?.prompt !== 'string' || payload.prompt === '') {
        throw new Error('prompt is required')
    }
    const attempt = payload.attempt
    return {
        sessionId: asString(payload.sessionId) ?? asString(headerSessionId) ?? randomUUID(),
        prompt: payload.prompt,
        eventId: asString(payload.eventId) ?? randomUUID(),
        attempt: Number.isInteger(attempt) && attempt > 0 ? attempt : 1,
    }
}
```

```ts
import {
    ENV_SESSION_ID, ENV_EVENT_ID, ENV_ATTEMPT, ENV_RECALL_RESULT,
    type LaunchSpec, type RoundRequest,
} from '@agentarts/harness-dsh-shared'

/** 精确白名单（spec §3.5）；DataLake 端点凭证变量名待对齐后由 P3 补充 */
const PASSTHROUGH = [
    'DSH_HOME', 'PATH', 'HOME', 'NODE_ENV', 'TZ', 'LANG',
    'DEEPSEEK_API_KEY', 'DEEPSEEK_BASE_URL',
]
/** 前缀白名单：AGENTRUN_* 与 MCP_*_URL */
const PREFIXES = ['AGENTRUN_', 'MCP_']

/**
 * 最小可运行版 LaunchSpec（spec §3.2）。C 模块（P3）会扩充白名单与凭证逻辑，
 * 此处保证 A+B 独立可运行。不全量继承 Adapter 环境——召回凭证不进子进程。
 */
export function buildLaunchSpec(req: RoundRequest, recallResult: string): LaunchSpec {
    const env: Record<string, string> = {}
    for (const [key, value] of Object.entries(process.env)) {
```

```ts
import { JsonResponse, type RequestContext } from '@huaweicloud/agentarts-sdk/sdk/runtime'
import type { Scheduler } from '../scheduler/index.js'
import { normalizeRequest } from '../request.js'
import { SESSION_HEADER } from '../types.js'

/**
 * A 模块：/invocations 处理器（spec §2.2）。只要 final 帧；
 * 响应头显式携带 sessionId（不依赖 SDK 从请求头回填）。
 */
export function createInvocationHandler(scheduler: Scheduler) {
    return async (payload: any, context?: RequestContext): Promise<JsonResponse> => {
        let req
        try {
            req = normalizeRequest(payload, context?.sessionId)
        } catch (err) {
            return new JsonResponse({ statusCode: 400, content: { error: (err as Error).message } })
        }
        const handle = await scheduler.run(req)
        const outcome = await handle.outcome
        const headers = { [SESSION_HEADER]: req.sessionId }
        if (!outcome.ok || !outcome.final) {
            return new JsonResponse({
                statusCode: 500,
                content: { error: outcome.error ?? 'no final frame' },
                headers,
            })
        }
        return new JsonResponse({
            content: { result: outcome.final.text, sessionId: req.sessionId },
            headers,
        })
    }
}
```

```ts
import {
    WebSocketDisconnect, type RequestContext, type WebSocketConnection,
} from '@huaweicloud/agentarts-sdk/sdk/runtime'
import type { Scheduler } from '../scheduler/index.js'
import { normalizeRequest } from '../request.js'

/**
 * A 模块：/ws 处理器（spec §3.3）。帧透传，一轮结束不关连接；
 * 对端断开静默退出。与 /ping 分离，长连接不影响探活。
 */
export function createWsHandler(scheduler: Scheduler) {
    return async (ws: WebSocketConnection, _ctx: RequestContext): Promise<void> => {
        while (!ws.closed) {
            let payload: any
            try {
                payload = await ws.receiveJson()
            } catch (err) {
                if (err instanceof WebSocketDisconnect) return
                throw err
            }
            let req
            try {
                req = normalizeRequest(payload)
            } catch (err) {
                await ws.sendJson({ type: 'error', message: (err as Error).message })
                continue
            }
            const handle = await scheduler.run(req)
            for await (const frame of handle.frames) {
                if (ws.closed) break
                await ws.sendJson(frame)
            }
            const outcome = await handle.outcome
            // 子进程失败但没来得及输出 error 帧时补发
            if (!outcome.ok && !ws.closed) {
                await ws.sendJson({ type: 'error', message: outcome.error ?? 'round failed' })
            }
        }
    }
}
```

```ts
import { randomUUID } from 'node:crypto'
import type { Frame } from '@agentarts/harness-dsh-shared'
import type { Scheduler } from '../scheduler/index.js'

/** 归一化 A2A 输入（SDK 侧负责把 A2A message 适配成此形状，契约待确认，spec §7 第 3 项） */
export interface A2ARequest {
    taskId: string
    sessionId: string
    prompt: string
    eventId: string
    attempt: number
}

/** taskId ↔ sessionId 映射（spec §2.3）；执行期间绑定，本轮结束解绑 */
export class TaskRegistry {
    private map = new Map<string, string>()
    bind(taskId: string, sessionId: string): void { this.map.set(taskId, sessionId) }
    lookup(taskId: string): string | undefined { return this.map.get(taskId) }
    unbind(taskId: string): void { this.map.delete(taskId) }
}

/**
 * A 模块：A2A 处理器。产出帧流，task 状态迁移/产物翻译在 SDK 的 A2A 面。
 * question 帧由 SDK 译为"需要输入"；用户答复经同一 sessionId 的下一轮进来。
 */
export function createA2AHandler(scheduler: Scheduler, registry: TaskRegistry) {
    return async function* (input: A2ARequest): AsyncIterable<Frame> {
        registry.bind(input.taskId, input.sessionId)
        try {
            const handle = await scheduler.run({
                sessionId: input.sessionId,
                prompt: input.prompt,
                eventId: input.eventId || randomUUID(),
                attempt: input.attempt ?? 1,
            })
            yield* handle.frames
            const outcome = await handle.outcome
            if (!outcome.ok) {
                yield { type: 'error', message: outcome.error ?? 'round failed' }
            }
        } finally {
            registry.unbind(input.taskId)
        }
    }
}
```

- [ ] **Step 3: 实现 `adapter/src/handlers/register.ts` 并接入 main.ts**

```ts
import type { Scheduler } from '../scheduler/index.js'
import { TaskRegistry, createA2AHandler } from './a2a.js'
import { createAguiHandler } from './agui.js'

/** SDK 新面的最小形状约定（spec §1.4 决策 8：app.a2a/app.agui 风格注册） */
interface ProtocolSurface {
    a2a?: (handler: unknown) => void
    agui?: (handler: unknown) => void
}

/**
 * 守卫注册：SDK 的 A2A/AG-UI 面 11/30 交付，未到位时跳过并告警，
 * 三端点不受影响。SDK 到位后若 API 形状不同，只需调整本文件。
 */
export function registerProtocolFaces(
    app: unknown, scheduler: Scheduler, registry: TaskRegistry,
): void {
    const surface = app as ProtocolSurface
    if (typeof surface.a2a === 'function') {
        surface.a2a(createA2AHandler(scheduler, registry))
    } else {
        console.warn('[adapter] SDK a2a 面未交付，跳过注册（交付后自动生效）')
    }
    if (typeof surface.agui === 'function') {
        surface.agui(createAguiHandler(scheduler))
    } else {
        console.warn('[adapter] SDK agui 面未交付，跳过注册（交付后自动生效）')
    }
}
```

main.ts 的 createApp 内、`app.websocket(...)` 之后加：

```ts

```

并补导入 `import { registerProtocolFaces } from './handlers/register.js'` 与
`import { TaskRegistry } from './handlers/a2a.js'`。

## 与既有目录结构（docs/design/目录结构.txt）的差异

1. **workspace 根**：txt 根标签为 `agentarts-adapter-js/`，实际仓库骨架在其下
   `deepseek-harness-adapter/` 一层——结构相同，本计划以实际骨架为准。
2. **不创建 `scheduler/semaphore.ts`**：txt 注释即"满则由 SDK 返回 503"，SDK
   `maxConcurrency` 已完整实现全局上限，自建即死代码（YAGNI）。
3. **新增文件**（txt 未列，均为 A+B 独立可运行所需）：
   - `scheduler/index.ts`：Scheduler 门面（queue + run + 端口组合）
   - `request.ts`：payload → RoundRequest 归一化
   - `launch-spec.ts`：最小 env 白名单 + spawn 参数；P3 扩充，届时可并入 config.ts
   - `handlers/register.ts`：SDK a2a/agui 面守卫注册
4. **shared 扩充**：txt 只列 `env.ts`；按已批准 spec §3.1/§3.5 增加 `frames.ts`、
   `launch.ts`、`index.ts`（帧契约与 LaunchSpec 是跨模块公共契约，放 shared）。
5. **config.ts / patch.ts / recall.ts / ops.ts 属 P3（C 模块）**，本计划不创建；
   main.ts 暂直接读 env（PORT/AGENTRUN_MAX_CONCURRENCY），P3 接入 config 后改装配线。
6. **跨仓链接修正（执行中发现）**：SDK 包内部依赖 `@huaweicloud/huaweicloud-sdk-agentidentity@workspace:*`，
   file: 链接后在本 workspace 无法解析——pnpm-workspace.yaml 增加 overrides 指向 SDK 仓
   本地包路径，并复制 SDK 仓的 packageExtensions（core 的 uuid 缺陷兜底）与 allowBuilds。
7. **测试串行执行（执行中发现）**：并行 vitest worker + 真实子进程 spawn 在本机触发
   V8 `VirtualAlloc failed` OOM——adapter/vitest.config.ts 设 `fileParallelism: false`、
   `maxWorkers: 1`；shared 套件无子进程，不受影响。
